# LSP-dita Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Sublime Text helper package that connects Sublime's LSP package to `dita-language-server`, adding client-side `href`/`conref` navigation and hover, a DITA-OT build, a root map picker, and format-on-save via LSP-lemminx.

**Architecture:** `boot.py` registers an `AbstractPlugin` subclass that downloads and launches the server jar. Six focused modules under `plugin/` each own one job. Every module splits a pure, testable core from a thin Sublime adapter, because the Sublime API cannot be imported in CI.

**Tech Stack:** Python 3.8 (Sublime Text's bundled runtime), Sublime Text 4 plugin API, the LSP package 2.13.0 plugin API, `unittest`, `ruff`, `mypy`.

**Spec:** `docs/superpowers/specs/2026-09-25-lsp-dita-design.md`

## Global Constraints

- Target Python 3.8. Sublime Text 4 bundles 3.8 for packages declaring it. No `match`, no walrus in public API, no `dict |` merge.
- Every module under `plugin/` must import cleanly without `sublime` present, or must confine `sublime` imports so that `tests/` can import the pure functions. Pure logic goes in modules that never import `sublime` at module scope.
- Server version pinned at `0.1.0`. Jar URL: `https://github.com/aireilly/dita-language-server/releases/download/v0.1.0/dita-language-server-0.1.0-all.jar`. SHA256: `e865acc51053b4158a4d7b200e97756433cb0477855f4476082489175f9d078c`.
- Minimum Java: 17.
- Package name string is `LSP-dita`. LSP session name is `dita`.
- Syntax scope is `text.xml.dita`.
- `ruff` line-length 120, rules `E,F,W`, matching `LSP-mdita`.
- Never bind Ctrl+S.

## Review Focus

Input classes the spec implies but no obvious task test covers. Each has its test assigned to the owning task.

1. A `.dita` file with two topics that both contain `id="step-1"` — fragment location must resolve within the addressed topic, not the first match in the file. (Task 4)
2. An `href` value containing a Windows-style backslash or a `..` traversal above the project root — resolution must normalise without escaping into unrelated directories. (Task 4)
3. A target file whose `<shortdesc>` contains nested markup such as `<ph>` or an undefined entity — extraction must return readable text rather than raising. (Task 5)
4. DITA-OT emitting an error line with no line/column (`Error: [DOTX008E] The resource ... cannot be loaded`) — the panel must still show it and the error count must include it. (Task 8)
5. `java -version` output from a JRE that prints `openjdk version "1.8.0_402"` — the legacy `1.x` version scheme must parse to major 8 and be rejected. (Task 2)

# plugin/java.py
import re
from typing import Optional

_VERSION_RE = re.compile(r'version\s+"(\d+)(?:\.(\d+))?')


def parse_java_major(version_output: str) -> Optional[int]:
    """Return the Java major version from `java -version` output, or None.

    Handles both the modern scheme ("17.0.20.1" -> 17) and the pre-9 scheme
    ("1.8.0_402" -> 8), where the leading 1 is a placeholder.
    """
    match = _VERSION_RE.search(version_output)
    if not match:
        return None
    first = int(match.group(1))
    if first == 1:
        second = match.group(2)
        return int(second) if second is not None else None
    return first
```

# plugin/server.py
import hashlib
import os
import shutil
import urllib.request

from .constants import JAR_NAME, JAR_SHA256, JAR_URL, SERVER_VERSION

_MARKER = "VERSION"


def needs_install(basedir: str, jar_name: str, version: str) -> bool:
    if not os.path.isfile(os.path.join(basedir, jar_name)):
        return True
    marker = os.path.join(basedir, _MARKER)
    if not os.path.isfile(marker):
        return True
    with open(marker, "r", encoding="utf-8") as f:
        return f.read().strip() != version


def verify_sha256(path: str, expected: str) -> bool:
    digest = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest().lower() == expected.strip().lower()


def download_jar(url: str, dest: str) -> None:
    with urllib.request.urlopen(url, timeout=120) as response:
        with open(dest, "wb") as out:
            shutil.copyfileobj(response, out)


def install(basedir: str) -> None:
    os.makedirs(basedir, exist_ok=True)
    final = os.path.join(basedir, JAR_NAME)
    partial = final + ".part"
    download_jar(JAR_URL, partial)
    if not verify_sha256(partial, JAR_SHA256):
        os.remove(partial)
        raise RuntimeError(
            "Checksum mismatch for {}. Expected sha256 {}.".format(JAR_NAME, JAR_SHA256)
        )
    os.replace(partial, final)
    with open(os.path.join(basedir, _MARKER), "w", encoding="utf-8") as f:
        f.write(SERVER_VERSION)
```

```python
# plugin/refs.py
import os
import re
from typing import NamedTuple, Optional, Tuple

RESOLVABLE_ATTRIBUTES = ("href", "conref", "conrefend", "copy-to")

_URI_SCHEME_RE = re.compile(r"^[A-Za-z][A-Za-z0-9+.-]*:")
_ATTR_RE = re.compile(r'([A-Za-z_][\w.:-]*)\s*=\s*"([^"]*)"')


class Reference(NamedTuple):
    raw: str
    path: str
    topic_id: str
    element_id: str
    attribute: str


def parse_address(value: str) -> Tuple[str, str, str]:
    """Split a DITA address into (path, topic_id, element_id)."""
    value = value.strip()
    path, _, fragment = value.partition("#")
    topic_id, _, element_id = fragment.partition("/")
    return path, topic_id, element_id


def _element_span(text: str, offset: int) -> Tuple[int, int]:
    start = text.rfind("<", 0, offset + 1)
    if start == -1:
        start = 0
    end = text.find(">", offset)
    if end == -1:
        end = len(text)
    return start, end + 1


def reference_at(text: str, offset: int) -> Optional[Reference]:
    """Return the resolvable Reference whose attribute value contains offset."""
    start, end = _element_span(text, offset)
    element = text[start:end]
    local = offset - start
    found = None
    attributes = {}
    for match in _ATTR_RE.finditer(element):
        attributes[match.group(1)] = match.group(2)
        if match.start(2) <= local <= match.end(2):
            found = (match.group(1), match.group(2))
    if found is None:
        return None
    name, value = found
    if name not in RESOLVABLE_ATTRIBUTES:
        return None
    if attributes.get("scope") == "external" or attributes.get("format") == "html":
        return None
    if _URI_SCHEME_RE.match(value.strip()):
        return None
    path, topic_id, element_id = parse_address(value)
    if not path and not topic_id:
        return None
    return Reference(raw=value, path=path, topic_id=topic_id,
                     element_id=element_id, attribute=name)


def resolve_path(reference: Reference, current_file: str) -> Optional[str]:
    """Resolve a Reference to an absolute filesystem path."""
    if not current_file:
        return None
    if not reference.path:
        return current_file
    relative = reference.path.replace("\\", "/")
    base = os.path.dirname(current_file)
    return os.path.normpath(os.path.join(base, relative))
```

```python
# plugin/topics.py
import os
import re
from typing import Dict, NamedTuple, Optional, Tuple

_TAG_RE = re.compile(r"<[^>]*>")
_WS_RE = re.compile(r"\s+")
_TITLE_RE = re.compile(r"<title\b[^>]*>(.*?)</title>", re.DOTALL)
_SHORTDESC_RE = re.compile(r"<shortdesc\b[^>]*>(.*?)</shortdesc>", re.DOTALL)
_ENTITIES = (("&lt;", "<"), ("&gt;", ">"), ("&quot;", '"'), ("&apos;", "'"), ("&amp;", "&"))

_cache = {}  # type: Dict[str, Tuple[float, Optional[TopicInfo]]]


class TopicInfo(NamedTuple):
    title: str
    shortdesc: str


def strip_markup(fragment: str) -> str:
    """Flatten a markup fragment to readable text, leaving unknown entities intact."""
    text = _TAG_RE.sub("", fragment)
    for entity, char in _ENTITIES:
        text = text.replace(entity, char)
    return _WS_RE.sub(" ", text).strip()


def _id_pattern(value: str) -> "re.Pattern":
    return re.compile(r"""<([\w.:-]+)\b[^>]*\bid\s*=\s*["']{}["']""".format(re.escape(value)))


def find_fragment_offset(text: str, topic_id: str, element_id: str) -> Optional[int]:
    """Return the offset of the addressed element, scoped to the addressed topic."""
    if not topic_id:
        return None
    topic_match = _id_pattern(topic_id).search(text)
    if not topic_match:
        return None
    if not element_id:
        return topic_match.start()
    tag = topic_match.group(1)
    closing = text.find("</{}>".format(tag), topic_match.end())
    extent_end = len(text) if closing == -1 else closing
    element_match = _id_pattern(element_id).search(text, topic_match.end(), extent_end)
    if not element_match:
        return topic_match.start()
    return element_match.start()


def read_topic_info(path: str) -> Optional[TopicInfo]:
    """Read title and shortdesc from a DITA file, cached on (path, mtime)."""
    try:
        mtime = os.path.getmtime(path)
    except OSError:
        return None
    cached = _cache.get(path)
    if cached is not None and cached[0] == mtime:
        return cached[1]
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            text = f.read()
    except OSError:
        return None
    title_match = _TITLE_RE.search(text)
    shortdesc_match = _SHORTDESC_RE.search(text)
    info = TopicInfo(
        title=strip_markup(title_match.group(1)) if title_match else "",
        shortdesc=strip_markup(shortdesc_match.group(1)) if shortdesc_match else "",
    )
    _cache[path] = (mtime, info)
    return info
```

```python
# plugin/rootmap.py
import os
from typing import List, Optional

EXCLUDED_DIRS = frozenset(("out", "temp", "build", ".git", "node_modules", ".svn"))


def discover_maps(folders: List[str], limit: int = 200) -> List[str]:
    """Return sorted .ditamap paths under folders, skipping build output."""
    found = []  # type: List[str]
    for folder in folders:
        for dirpath, dirnames, filenames in os.walk(folder):
            dirnames[:] = [d for d in dirnames if d not in EXCLUDED_DIRS]
            for name in filenames:
                if name.endswith(".ditamap"):
                    found.append(os.path.join(dirpath, name))
                    if len(found) >= limit:
                        return sorted(found)
    return sorted(found)


def nearest_map(start_file: str, stop_at: str = "") -> Optional[str]:
    """Walk upwards from start_file looking for a .ditamap sibling."""
    current = os.path.dirname(os.path.abspath(start_file))
    while True:
        try:
            names = sorted(n for n in os.listdir(current) if n.endswith(".ditamap"))
        except OSError:
            names = []
        if names:
            return os.path.join(current, names[0])
        if stop_at and os.path.abspath(current) == os.path.abspath(stop_at):
            return None
        parent = os.path.dirname(current)
        if parent == current:
            return None
        current = parent
```

# plugin/build.py
import os
import re
import subprocess
import threading
import webbrowser
from typing import Iterable, List, Optional, Tuple

RESULT_FILE_REGEX = r"^(?:Error|Warning|Fatal|Info):\s+file:(/[^\s:]+):(\d+):(\d+):\s*(.*)$"

_ERROR_RE = re.compile(r"^(?:Error|Fatal):")
_WARNING_RE = re.compile(r"^Warning:")


def build_argv(dita_bin: str, map_path: str, transtype: str,
               output: str, extra_args: List[str]) -> List[str]:
    return [dita_bin, "-i", map_path, "-f", transtype, "-o", output] + list(extra_args)


def resolve_output(map_path: str, output: str) -> str:
    target = output or "out"
    if os.path.isabs(target):
        return os.path.normpath(target)
    return os.path.normpath(os.path.join(os.path.dirname(map_path), target))


def count_problems(lines: Iterable[str]) -> Tuple[int, int]:
    errors = 0
    warnings = 0
    for line in lines:
        if _ERROR_RE.match(line):
            errors += 1
        elif _WARNING_RE.match(line):
            warnings += 1
    return errors, warnings
